# Gleam Client Contract Tests Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Beryl-owned Gleam integration tests that exercise a real Beryl+Mist WebSocket server through Aquamarine and Gluegun clients.

**Architecture:** The suite starts an in-process Beryl coordinator and Mist WebSocket server on a dynamic localhost port, then uses Aquamarine with the Phoenix codec for normal channel scenarios. A small test-only Erlang FFI module allocates local ports, and direct Gluegun raw WebSocket use is reserved for malformed-frame coverage.

**Tech Stack:** Gleam Erlang target, Gleeunit, Beryl, Mist, Aquamarine, Gluegun, Erlang FFI.

## Global Constraints

- Add Beryl-owned integration tests that exercise a real Beryl server over WebSocket from Gleam.
- Use Aquamarine as the normal client API for join, push, receive, close, and heartbeat scenarios.
- Use Gluegun only for raw malformed-frame cases that Aquamarine should not expose.
- Keep tests isolated with dynamic localhost ports, bounded receives, explicit cleanup, and mailbox discipline.
- Assert server behavior at the contract level, not Aquamarine implementation details.
- Do not rebuild a Phoenix channel client inside Beryl tests.
- Do not add browser, Playwright, or JavaScript-target coverage for this suite.
- If the suite needs normal client behavior that Aquamarine does not expose yet, file an issue in `tylerbutler/aquamarine` and keep the Beryl test focused on the currently available contract.
- Update `gleam.toml` and `manifest.toml` through Gleam tooling so dependency metadata stays consistent.
- Use `gleam test -- --filter "test_name"` for focused tests and `just test` for the final suite.

fn start_test_server(
  register: fn(beryl.Channels, process.Subject(TestEvent)) -> Nil,
  events: process.Subject(TestEvent),
) -> Result(TestServer, Nil) {
  let assert Ok(port) = available_port()
  let assert Ok(channels) = beryl.start(beryl.config(wire.phoenix_codec()))
  register(channels, events)

  let handler = fn(req) {
    mist_transport.upgrade(
      req,
      channels,
      mist_transport.default_config(socket_path),
      fn() {
        response.new(404)
        |> response.set_body(mist.Bytes(bytes_tree.new()))
      },
    )
  }

  case
    mist.new(handler)
    |> mist.bind("127.0.0.1")
    |> mist.port(port)
    |> mist.start
  {
    Ok(supervisor) -> Ok(TestServer(channels, port, supervisor))
    Error(_) -> Error(Nil)
  }
}

fn stop_test_server(server: TestServer) -> Nil {
  process.send_exit(to: server.supervisor.pid)
}

fn register_lobby(channels: beryl.Channels) -> Nil {
  let lobby =

        socket: socket,
      )
    })
  let assert Ok(_) = beryl.register(channels, "test:lobby", lobby)
  Nil
}

```erlang
-module(beryl_test_port_ffi).
-export([available_port/0]).

available_port() ->
    case gen_tcp:listen(0, [
        inet,
        {ip, {127, 0, 0, 1}},
        {active, false},
        {reuseaddr, false}
    ]) of
        {ok, Socket} ->
            Result = case inet:sockname(Socket) of
                {ok, {{127, 0, 0, 1}, Port}} -> {ok, Port};
                {ok, {_Address, Port}} -> {ok, Port};
                {error, _Reason} -> {error, nil}
            end,
            _ = gen_tcp:close(Socket),
            Result;
        {error, _Reason} ->
            {error, nil}
    end.
```

fn receive_joined(
  events: process.Subject(TestEvent),
  topic: String,
) -> Result(Nil, Nil) {
  let selector =
    process.new_selector()
    |> process.select(events)

  case process.selector_receive(selector, 500) {
    Ok(Joined(joined_topic)) if joined_topic == topic -> Ok(Nil)
    _ -> Error(Nil)
  }
}

fn receive_terminated(
  events: process.Subject(TestEvent),
  topic: String,
) -> Result(bchannel.StopReason, Nil) {
  let selector =
    process.new_selector()
    |> process.select(events)

  case process.selector_receive(selector, 500) {
    Ok(Terminated(stopped_topic, reason)) if stopped_topic == topic ->
      Ok(reason)
    _ -> Error(Nil)
  }
}
