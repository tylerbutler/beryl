This guide walks you through building a working real-time app from scratch:
a Gleam server with app-side dispatch, wired to a Phoenix JS client running in
the browser.

The snippets here are abbreviated to keep things readable. For a fully
runnable application with HTML, static assets, and end-to-end tests, see the
[examples](/examples/).

## Prerequisites

- Gleam project targeting **Erlang** (`gleam new my_app`)
- `gleam add beryl beryl_mist` adds beryl, the Mist WebSocket transport, and all their dependencies

## 1. Define your socket app

With beryl's current API, you start one socket runtime and provide two
functions:

- `init` runs once per connection and builds your per-socket `Model`
- `update` receives every `Join`, `Message`, `Binary`, `Closed`, and `Info` event

```gleam
// src/my_app/room_app.gleam
import beryl/event
import beryl/wire
import gleam/json

pub type Msg {
  NoOp
}

pub type Model {
  Model(socket_id: String)
}

pub fn init(info: event.ConnectInfo(Msg)) -> #(Model, List(event.Effect)) {
  #(Model(socket_id: info.socket_id), [])
}

pub fn update(model: Model, ev: event.Input(Msg)) -> event.Next(Model, Msg) {
  case ev {
    event.Join("room:" <> _, payload, ref) ->
      event.Next(
        model,
        [event.AcceptJoin(ref, Some(wire.dynamic_to_json(payload)))],
      )

    event.Message(topic, "new_msg", payload, _ref) ->
      event.Next(
        model,
        [event.Broadcast(topic, "new_msg", wire.dynamic_to_json(payload))],
      )

    event.Message(_, _, _, Some(ref)) ->
      event.Next(
        model,
        [event.ReplyError(
          ref,
          json.object([#("error", json.string("unknown_event"))]),
        )],
      )

    event.Binary(_, _)
    | event.Closed(_, _)
    | event.Info(_)
    | event.Message(_, _, _, None) ->
      event.Next(model, [])
  }
}
```

<Aside type="note" title="Reply vs Push vs Broadcast">
- **`event.ReplyOk` / `event.ReplyError`** — send `phx_reply` back to the calling client on the original ref.
- **`event.Push`** — send a named event to this socket only, on a topic it already joined.
- **`event.Broadcast`** — send a named event to every socket on the topic, including the sender.
- **`event.Next(model, [])`** — continue with no outgoing frames.
</Aside>

## 2. Start beryl and wire the transport

Now start the socket runtime and pass the returned `beryl.Sockets` handle to the
Mist transport:

```gleam
// src/my_app.gleam
import beryl
import beryl_mist as mist_transport
import beryl/wire
import gleam/bytes_tree
import gleam/erlang/process
import gleam/http/request
import gleam/http/response
import mist
import my_app/room_app

pub fn main() {
  let config =
    beryl.config(wire.phoenix_codec())
    |> beryl.with_message_rate(per_second: 30, burst: 60)

  let assert Ok(sockets) =
    beryl.start(config, init: room_app.init, update: room_app.update)

  let assert Ok(_) =
    fn(req) {
      mist_transport.upgrade(
        req,
        sockets,
        mist_transport.default_config("/socket/websocket"),
        fn() { handle_http(req) },
      )
    }
    |> mist.new
    |> mist.port(8000)
    |> mist.start

  process.sleep_forever()
}

fn handle_http(
  req: request.Request(mist.Connection),
) -> response.Response(mist.ResponseData) {
  case request.path_segments(req) {
    [] ->
      response.new(200)
      |> response.set_body(mist.Bytes(bytes_tree.from_string("Hello!")))
    _ ->
      response.new(404)
      |> response.set_body(mist.Bytes(bytes_tree.new()))
  }
}
```

<Aside type="tip" title="Transport path">
`mist_transport.default_config("/socket/websocket")` tells beryl to intercept
requests on `/socket/websocket`. When you create a Phoenix JS socket with
`new Socket("/socket")`, the client appends `/websocket` automatically — so
the paths line up.
</Aside>

## 3. Connect from the browser

beryl uses the same wire format as Phoenix channels, so you can use the official
`phoenix` npm package (or CDN build) in your frontend.

<Tabs>
  <TabItem label="Install (npm)">
```bash
npm install phoenix
```
  </TabItem>
  <TabItem label="CDN">
```html
<script src="https://cdn.jsdelivr.net/npm/phoenix/priv/static/phoenix.min.js"></script>
```
  </TabItem>
</Tabs>

```js
import { Socket } from "phoenix";

// "/socket" → client appends "/websocket" → hits "/socket/websocket" on the server.
const socket = new Socket("/socket");
socket.connect();

// Join the "room:lobby" topic.
const channel = socket.channel("room:lobby", { username: "alice" });

channel
  .join()
  .receive("ok", (resp) => {
    // resp is the JSON payload you returned from AcceptJoin.
    console.log("Joined as", resp.username);
  })
  .receive("error", (resp) => {
    console.error("Join failed", resp);
  });

// Listen for broadcast messages.
channel.on("new_msg", (payload) => {
  console.log("Message:", payload);
});

// Send a message.
channel.push("new_msg", { text: "Hello, world!" });
```

## 4. Acknowledging a specific push

If you want the server to confirm an individual client push, match the message
when a ref is present and return `event.ReplyOk`:

```gleam
event.Message(topic, "new_msg", payload, Some(ref)) ->
  event.Next(
    model,
    [
      event.Broadcast(topic, "new_msg", wire.dynamic_to_json(payload)),
      event.ReplyOk(
        ref,
        json.object([#("status", json.string("ok"))]),
      ),
    ],
  )
```

```js
channel
  .push("new_msg", { text: "Hello!" })
  .receive("ok", (resp) => {
    // resp is the payload from ReplyOk — { status: "ok" }
    console.log("Delivered", resp);
  });
```

## 5. Rejecting a join

Return `event.RejectJoin` when a topic should not be admitted:

```gleam
event.Join(topic, _payload, ref) ->
  case is_room_valid(topic) {
    False ->
      event.Next(
        model,
        [event.RejectJoin(
          ref,
          reason: json.object([#("reason", json.string("room_not_found"))]),
        )],
      )

    True ->
      event.Next(model, [event.AcceptJoin(ref, None)])
  }
```

A `Join` that reaches the end of the update turn unanswered is also rejected
automatically, so joins fail closed by default.
