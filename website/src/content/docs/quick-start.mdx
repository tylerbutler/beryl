This guide walks you through building a working real-time channel from scratch:
a Gleam server that handles join and message events, wired to a Phoenix JS client
running in the browser.

The snippets here are abbreviated to keep things readable. For a fully
runnable application with HTML, static assets, and end-to-end tests, see the
[examples](/examples/).

## Prerequisites

- Gleam project targeting **Erlang** (`gleam new my_app`)
- `gleam add beryl beryl_mist` adds beryl, the Mist WebSocket transport, and all their dependencies

## 1. Define your app

A beryl app is two functions. `init` builds a per-socket **model** when a
client connects; `update` receives every **event** for that socket and
returns the next model plus a list of **effects**.

```gleam
// src/my_app/room_app.gleam
import beryl/event.{
  type Event, type Next, AcceptJoin, Broadcast, Join, Message, Next,
}
import gleam/dynamic.{type Dynamic}
import gleam/dynamic/decode
import gleam/json
import gleam/option.{Some}

/// Per-socket state — anything you want to remember about this connection.
pub type Model {
  Model(username: String)
}

pub fn init(_info: event.ConnectInfo(Nil)) -> #(Model, List(event.Effect)) {
  #(Model(username: "Anonymous"), [])
}

pub fn update(model: Model, ev: Event(Nil)) -> Next(Model, Nil) {
  case ev {
    // A client asked to join a "room:*" topic.
    Join("room:" <> _, payload, ref) -> {
      // Payloads arrive as `Dynamic`; decode the fields you need.
      let username = string_field(payload, "username", or: "Anonymous")

      // AcceptJoin answers the join; the reply is delivered to the client
      // as phx_reply on the join ref.
      Next(Model(username: username), [
        AcceptJoin(
          ref,
          Some(json.object([#("username", json.string(username))])),
        ),
      ])
    }
    // Reject joins for topics you don't serve.
    Join(_, _, ref) ->
      Next(model, [
        event.RejectJoin(
          ref,
          json.object([#("reason", json.string("unknown_topic"))]),
        ),
      ])

    // A client push on a joined topic.
    Message(topic, "new_msg", payload, _ref) -> {
      let text = string_field(payload, "text", or: "")
      // Broadcast to every socket joined to this topic, including the
      // sender.
      Next(model, [
        Broadcast(
          topic,
          "new_msg",
          json.object([
            #("username", json.string(model.username)),
            #("text", json.string(text)),
          ]),
        ),
      ])
    }

    _ -> Next(model, [])
  }
}

fn string_field(payload: Dynamic, name: String, or default: String) -> String {
  case
    decode.run(payload, {
      use value <- decode.field(name, decode.string)
      decode.success(value)
    })
  {
    Ok(value) -> value
    Error(_) -> default
  }
}
```

<Aside type="note" title="Reply vs Push vs Broadcast">
- **`ReplyOk(ref, payload)`** — sends a `phx_reply` back to the *calling*
  client tied to the original message ref (available when the event's `ref`
  is `Some`). Use it to acknowledge a specific push.
- **`Push(topic, event, payload)`** — a server-initiated message to this
  socket only.
- **`Broadcast(topic, event, payload)`** — a named event to *all* sockets on
  the topic, including the sender. Use `BroadcastFrom` to exclude the sender.
- Returning no effects silently continues — nothing is sent to the caller.
</Aside>

## 2. Start beryl and the transport

Wire everything together in your application entry point:

```gleam
// src/my_app.gleam
import beryl
import beryl_mist as mist_transport
import beryl/wire
import gleam/bytes_tree
import gleam/erlang/process
import gleam/http/request
import gleam/http/response
import gleam/otp/static_supervisor
import mist
import my_app/room_app

pub fn main() {
  // Start the beryl app-dispatch system.
  let assert Ok(#(channels, spec)) =
    beryl.child_spec(
      beryl.config(wire.phoenix_codec()),
      init: room_app.init,
      update: room_app.update,
    )
  let assert Ok(_root) =
    static_supervisor.new(static_supervisor.OneForOne)
    |> static_supervisor.add(spec)
    |> static_supervisor.start()

  // Start the HTTP + WebSocket server.
  let assert Ok(_) =
    fn(req) {
      // Upgrade WebSocket requests first; fall through to HTTP for everything else.
      mist_transport.upgrade(
        req,
        channels,
        mist_transport.default_config("/socket/websocket"),
        fn() { handle_http(req) },
      )
    }
    |> mist.new
    |> mist.port(8000)
    |> mist.start

  process.sleep_forever()
}

fn handle_http(
  req: request.Request(mist.Connection),
) -> response.Response(mist.ResponseData) {
  case request.path_segments(req) {
    [] ->
      response.new(200)
      |> response.set_body(mist.Bytes(bytes_tree.from_string("Hello!")))
    _ ->
      response.new(404)
      |> response.set_body(mist.Bytes(bytes_tree.new()))
  }
}
```

<Aside type="tip" title="Transport path">
`mist_transport.default_config("/socket/websocket")` tells beryl to intercept
requests on `/socket/websocket`. When you create a Phoenix JS socket with
`new Socket("/socket")`, the client appends `/websocket` automatically — so
the paths line up.
</Aside>

## 3. Connect from the browser

beryl uses the same wire format as Phoenix channels, so you can use the official
`phoenix` npm package (or CDN build) in your frontend.

<Tabs>
  <TabItem label="Install (npm)">
```bash
npm install phoenix
```
  </TabItem>
  <TabItem label="CDN">
```html
<script src="https://cdn.jsdelivr.net/npm/phoenix/priv/static/phoenix.min.js"></script>
```
  </TabItem>
</Tabs>

```js
import { Socket } from "phoenix";

// "/socket" → client appends "/websocket" → hits "/socket/websocket" on the server.
const socket = new Socket("/socket");
socket.connect();

// Join the "room:lobby" topic.
const channel = socket.channel("room:lobby", { username: "alice" });

channel
  .join()
  .receive("ok", (resp) => {
    // resp is the AcceptJoin reply payload — { username: "alice" }
    console.log("Joined as", resp.username);
  })
  .receive("error", (resp) => {
    console.error("Join failed", resp);
  });

// Listen for broadcast messages.
channel.on("new_msg", (payload) => {
  console.log("Message:", payload);
});

// Send a message.
channel.push("new_msg", { text: "Hello, world!" });
```

## 4. Acknowledging a specific push

If you want the server to confirm delivery of an individual push, return a
`ReplyOk` effect for the message's ref. The client receives a `phx_reply`
event tied to the original message ref:

```gleam
// Server side — acknowledge message delivery
Message(topic, "new_msg", payload, Some(ref)) ->
  Next(model, [
    Broadcast(
      topic,
      "new_msg",
      json.object([#("username", json.string(model.username))]),
    ),
    event.ReplyOk(ref, json.object([#("status", json.string("ok"))])),
  ])
```

```js
// Client side — receive the acknowledgment
channel
  .push("new_msg", { text: "Hello!" })
  .receive("ok", (resp) => {
    // resp is the ReplyOk payload — { status: "ok" }
    console.log("Delivered", resp);
  });
```

The `ref` is `Some` whenever the client expects a reply; a `Message` whose
`ref` is `None` cannot be answered (there is nothing to correlate a reply
with).

## 5. Rejecting a join

Return a `RejectJoin` effect to refuse a client:

```gleam
Join("room:" <> room_id, _payload, ref) ->
  case is_room_valid(room_id) {
    False ->
      Next(model, [
        event.RejectJoin(
          ref,
          json.object([#("reason", json.string("Room not found"))]),
        ),
      ])
    True -> // ... AcceptJoin as normal
  }
```

The client receives an `"error"` reply on its `.join()` call. A `Join` you
leave unanswered is rejected automatically — beryl fails closed.
